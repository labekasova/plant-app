// Хук для работы с LocalStorage
function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(error);
      return initialValue;
    }
  });

  const setValue = (value) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.error(error);
    }
  };
  return [storedValue, setValue];
}

const Button = ({ onClick, children, variant = 'primary', className = '', ...props }) => {
  const variants = {
    primary: 'bg-green-500 hover:bg-green-400 border-green-700 text-white',
    secondary: 'bg-amber-200 hover:bg-amber-100 border-amber-400 text-amber-900',
    danger: 'bg-red-500 hover:bg-red-400 border-red-700 text-white',
    water: 'bg-blue-500 hover:bg-blue-400 border-blue-700 text-white'
  };
  
  return (
    <button 
      onClick={onClick}
      className={`game-font px-4 py-2 rounded-xl border-b-4 active:border-b-0 active:translate-y-1 transition-all ${variants[variant]} ${className}`}
      {...props}
    >
      {children}
    </button>
  );
};

const PlantPot = ({ plant, onClick }) => {
  // Расчет жажды. В реальном приложении это дни. Для прототипа 1 день = 24 часа.
  const msInDay = 1000 * 60 * 60 * 24;
  const daysSinceWatered = (Date.now() - plant.lastWatered) / msInDay;
  const isThirsty = daysSinceWatered >= plant.waterIntervalDays;
  
  // Добавляем рандомную задержку анимации, чтобы они дышали вразнобой
  const animDelay = `${(plant.id.charCodeAt(0) % 10) * 0.2}s`;

  return (
    <div 
      className="relative flex flex-col items-center cursor-pointer group w-20 sm:w-24 mt-auto mb-1 transition-transform hover:scale-105"
      onClick={() => onClick(plant)}
    >
      {/* Иконка нужды в воде */}
      {isThirsty && (
        <div className="absolute -top-10 z-30 text-3xl water-drop drop-shadow-md">
          💧
        </div>
      )}

      {/* Само растение */}
      <div 
        className="text-[60px] sm:text-[80px] leading-none z-10 plant-breathe drop-shadow-xl relative"
        style={{ animationDelay: animDelay }}
      >
        {PLANT_TYPES[plant.type].icon}
        {/* Затемнение, если сильно засохло (прошло в 1.5 раза больше времени) */}
        {daysSinceWatered > (plant.waterIntervalDays * 1.5) && (
          <div className="absolute inset-0 bg-black/20 rounded-full blur-sm z-20 pointer-events-none mix-blend-overlay"></div>
        )}
      </div>

      {/* Горшок */}
      <div className="w-16 sm:w-20 h-12 sm:h-14 bg-orange-700 rounded-b-xl border-t-[6px] border-orange-900 flex items-center justify-center shadow-lg relative z-20 -mt-2">
        <div className="absolute inset-0 bg-gradient-to-b from-transparent to-black/20 rounded-b-xl"></div>
        {/* Табличка с именем и локацией */}
        <div className="flex items-center gap-1 bg-amber-950 px-2 py-0.5 rounded border border-amber-900/50 shadow-inner max-w-[90%] relative z-10">
          {plant.location === 'outdoor' ? (
            <span className="text-[8px] sm:text-[10px]">☀️</span>
          ) : (
            <span className="text-[8px] sm:text-[10px]">🏠</span>
          )}
          <div className="game-font text-[10px] sm:text-[12px] text-white truncate">
            {plant.name}
          </div>
        </div>
      </div>
      
      {/* Тень на полке */}
      <div className="w-14 sm:w-16 h-3 bg-black/40 rounded-full -mt-2 blur-[3px]"></div>
    </div>
  );
};

  const [isAddModalOpen, setIsAddModalOpen] = useState(false);
  const [selectedShelf, setSelectedShelf] = useState(null);
  const [editingPlant, setEditingPlant] = useState(null);

  const t = (key) => TRANSLATIONS[lang][key] || key;

  // Форма добавления/редактирования
  const [formData, setFormData] = useState({
    name: '',
    type: 'succulent',
    waterIntervalDays: 3,
    notes: '',
    location: 'indoor'
  });

  const openAddModal = (shelfIndex) => {
    setSelectedShelf(shelfIndex);
    setFormData({ name: '', type: 'succulent', waterIntervalDays: 3, notes: '', location: 'indoor' });
    setIsAddModalOpen(true);
  };

  const handleSavePlant = () => {
    if (!formData.name.trim()) return;

    if (editingPlant) {
      // Обновление
      setPlants(plants.map(p => p.id === editingPlant.id ? { ...p, ...formData } : p));
      setEditingPlant(null);
    } else {
      // Добавление
      const newPlant = {
        ...formData,
        id: Math.random().toString(36).substr(2, 9),
        shelfIndex: selectedShelf,
        lastWatered: Date.now(), // Полито при посадке
      };
      setPlants([...plants, newPlant]);
      setIsAddModalOpen(false);
    }
  };

  const handleWater = (id) => {
    setPlants(plants.map(p => p.id === id ? { ...p, lastWatered: Date.now() } : p));
  };

  const handleDelete = (id) => {
    setPlants(plants.filter(p => p.id !== id));
    setEditingPlant(null);
  };

  // Функция для тестов: состарить растение, чтобы оно захотело пить
  const debugMakeThirsty = (id, intervalDays) => {
    const msInDay = 1000 * 60 * 60 * 24;
    const oldDate = Date.now() - (intervalDays * msInDay + 1000); // на 1 секунду больше нужного интервала
    setPlants(plants.map(p => p.id === id ? { ...p, lastWatered: oldDate } : p));
  };

  return (
    <div className="min-h-screen pb-20 pt-8 relative">
      {/* Солнце (декорация) */}
      <div className="absolute top-8 right-8 w-24 h-24 bg-yellow-300 rounded-full opacity-80 blur-lg mix-blend-screen animate-pulse pointer-events-none"></div>
      <div className="absolute top-12 right-12 w-16 h-16 bg-yellow-100 rounded-full shadow-[0_0_50px_20px_rgba(253,224,71,0.5)] pointer-events-none"></div>

      {}
      <div className="absolute top-4 left-4 z-20 flex gap-2 bg-white/40 p-1 rounded-xl backdrop-blur-sm border border-white/50 shadow-sm">
        <button 
          onClick={() => setLang('ru')} 
          className={`px-3 py-1 rounded-lg font-bold transition-colors ${lang === 'ru' ? 'bg-green-500 text-white' : 'hover:bg-white/50 text-green-900'}`}
        >
          RU
        </button>
        <button 
          onClick={() => setLang('en')} 
          className={`px-3 py-1 rounded-lg font-bold transition-colors ${lang === 'en' ? 'bg-green-500 text-white' : 'hover:bg-white/50 text-green-900'}`}
        >
          EN
        </button>
      </div>

      <div className="max-w-3xl mx-auto px-4 relative z-10">
        
        {/* Заголовок */}
        <div className="text-center mb-12">
          <h1 className="text-4xl md:text-6xl game-font text-white drop-shadow-[0_4px_4px_rgba(0,0,0,0.5)] stroke-black stroke-2" 
              style={{ WebkitTextStroke: '2px #166534' }}>
            {t('title')}
          </h1>
          <p className="text-green-900 font-bold mt-2 bg-white/40 inline-block px-4 py-1 rounded-full backdrop-blur-sm border border-white/50">
            {t('subtitle')}
          </p>
        </div>

        {/* Сетка полок */}
        <div className="flex flex-col gap-16 md:gap-24">

            const shelfPlants = plants.filter(p => p.shelfIndex === shelfIndex);
            const isFull = shelfPlants.length >= MAX_PLANTS_PER_SHELF;
